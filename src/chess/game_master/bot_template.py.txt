    def choose_move(self, request: MoveRequest) -> str | None:
        # Return one UCI move from request.legal_moves, such as "e2e4" or "e7e8q".
        # Return None to resign. An exception also resigns, with a traceback in the bot log.
        # Answer within request.deadline_s seconds, or the game is lost on time.
        state = GameState.from_request(request, self.game.initial_fen)

        # position after a move, so you can look ahead without touching the real game.

        return self.rng.choice(captures or request.legal_moves)
